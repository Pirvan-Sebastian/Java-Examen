        for (Produs produs : listaProduse) 
        {
            
            // 4. Creăm un obiect JSON (reprezintă acoladele {})
            JSONObject obiectJson = new JSONObject();
            
            // 5. Adăugăm atributele sub formă de "Cheie" : Valoare
            obiectJson.put("cod_produs", produs.getCod());
            obiectJson.put("denumire", produs.getDenumire());
            obiectJson.put("pret", produs.getPret());
            
            // 6. Adăugăm obiectul în vectorul mare
            arrayJson.put(obiectJson);
        }
        
        // 7. Salvăm totul în fișierul text
        writer.write(arrayJson.toString(4));
        
        System.out.println("Fisierul JSON a fost generat cu succes!");
